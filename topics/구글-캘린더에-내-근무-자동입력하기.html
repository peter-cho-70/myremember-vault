🔄 자동화 Python 스크립트

import json
from datetime import datetime, timedelta

# Google Calendar API 설정
PERSONAL_CALENDAR_ID = "ubsoldboy@gmail.com"
SHIFT_TIMES = {
    'R': ('12:30', '19:30'),
    'L': ('06:00', '16:00'),
    'U': ('15:30', '24:00'),
    'W': ('00:00', '07:30')
}

def time_to_iso(date_str, time_str):
    """시간을 ISO 8601 형식으로 변환"""
    if time_str == '24:00':
        # 자정은 다음날 00:00으로
        date_obj = datetime.strptime(date_str, '%Y-%m-%d')
        date_obj += timedelta(days=1)
        return date_obj.strftime('%Y-%m-%dT00:00:00+09:00')
    
    return f"{date_str}T{time_str}:00+09:00"

def create_work_event(date_str, shift_type):
    """근무 이벤트 생성"""
    start_time, end_time = SHIFT_TIMES[shift_type]
    
    start_iso = time_to_iso(date_str, start_time)
    end_iso = time_to_iso(date_str, end_time)
    
    event = {
        "summary": "근무",
        "description": f"{shift_type} 근무",
        "start": {"dateTime": start_iso, "timeZone": "Asia/Seoul"},
        "end": {"dateTime": end_iso, "timeZone": "Asia/Seoul"}
    }
    
    return event

def add_bulk_events(schedule_list):
    """일괄 이벤트 추가
    
    Args:
        schedule_list: [('2026-09-01', 'R'), ('2026-09-02', 'L'), ...] 형식
    """
    events = []
    for date_str, shift_type in schedule_list:
        if shift_type in SHIFT_TIMES:
            event = create_work_event(date_str, shift_type)
            events.append(event)
    
    return events

# 사용 예시
schedule_from_calendar = [
    ('2026-09-01', 'L'),
    ('2026-09-02', 'U'),
    ('2026-09-03', 'W'),
    # ... 보도기술부 일정에서 추출한 모든 R, L, U, W
]

events = add_bulk_events(schedule_from_calendar)
print(f"생성된 이벤트: {len(events)}개")

3. 시간: 해당 근무 유형의 시작~종료 시간
4. 타임존: Asia/Seoul
